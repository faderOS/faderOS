cmake_minimum_required(VERSION 3.20)
# Keep CLI, LCD and package versions synchronized with the source definition.
file(STRINGS src/link/home.hpp host_version REGEX "HostVersion=")
string(REGEX MATCH "[0-9]+\\.[0-9]+\\.[0-9]+" host_version "${host_version}")
option(PICOHOST_BUILD_POSIX "Build the Linux serial, mixer and web runtime" ON)
if(PICOHOST_BUILD_POSIX)
  set(platform_default posix)
else()
  set(platform_default core)
endif()
set(FADEROS_PLATFORM "${platform_default}" CACHE STRING "Target platform: posix, pico or core")
set_property(CACHE FADEROS_PLATFORM PROPERTY STRINGS posix pico core)
if(FADEROS_PLATFORM STREQUAL "pico")
  set(PICOHOST_BUILD_POSIX OFF)
  if(NOT PICO_SDK_PATH AND DEFINED ENV{PICO_SDK_PATH})
    set(PICO_SDK_PATH "$ENV{PICO_SDK_PATH}")
  endif()
  if(NOT EXISTS "${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
    message(FATAL_ERROR "Set PICO_SDK_PATH to an installed Pico SDK (2.0 or newer).")
  endif()
  if(NOT PICO_BOARD)
    set(PICO_BOARD pico2 CACHE STRING "Pico SDK board")
  endif()
  include("${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
  project(faderOS VERSION ${host_version} LANGUAGES C CXX ASM)
  pico_sdk_init()
  if(PICO_SDK_VERSION_STRING VERSION_LESS 2.0)
    message(FATAL_ERROR "faderOS requires Pico SDK 2.0 or newer.")
  endif()
elseif(FADEROS_PLATFORM STREQUAL "posix" OR FADEROS_PLATFORM STREQUAL "core")
  project(faderOS VERSION ${host_version} LANGUAGES CXX)
  if(FADEROS_PLATFORM STREQUAL "core")
    set(PICOHOST_BUILD_POSIX OFF)
  elseif(NOT PICOHOST_BUILD_POSIX)
    message(FATAL_ERROR "Use FADEROS_PLATFORM=core when disabling the POSIX runtime.")
  endif()
else()
  message(FATAL_ERROR "Unknown FADEROS_PLATFORM: choose posix, pico or core.")
endif()
if(PICOHOST_BUILD_POSIX AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "The POSIX runtime currently supports Linux only.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
include(GNUInstallDirs)
add_compile_options(-Wall -Wextra $<$<COMPILE_LANGUAGE:CXX>:-Wpedantic>)
set(core_sources transitions keys config mixer core panel setup)
list(TRANSFORM core_sources PREPEND src/link/)
list(TRANSFORM core_sources APPEND .cpp)
add_library(picohost_core STATIC ${core_sources})
target_include_directories(picohost_core PUBLIC src)
target_compile_options(picohost_core PRIVATE -fno-exceptions -fno-rtti)
string(TOUPPER "${FADEROS_PLATFORM}" platform_define)
target_compile_definitions(picohost_core PUBLIC FADEROS_PLATFORM_${platform_define}=1)
if(FADEROS_PLATFORM STREQUAL "pico")
  set(FADEROS_UART_INDEX 0 CACHE STRING "Panel UART controller: 0 or 1")
  set(FADEROS_UART_TX 0 CACHE STRING "Panel UART TX GPIO (through MAX3232)")
  set(FADEROS_UART_RX 1 CACHE STRING "Panel UART RX GPIO (through MAX3232)")
  set(FADEROS_PANEL_BAUD 38400 CACHE STRING "Preferred panel speed: 9600 or 38400")
  if(NOT FADEROS_UART_INDEX MATCHES "^[01]$" OR NOT FADEROS_PANEL_BAUD MATCHES "^(9600|38400)$")
    message(FATAL_ERROR "Choose UART 0/1 and panel baud 9600/38400.")
  endif()
  add_executable(faderOS-pico src/platform/pico/main.cpp src/platform/pico/uart_transport.cpp)
  target_link_libraries(faderOS-pico PRIVATE picohost_core pico_stdlib hardware_uart hardware_sync)
  target_compile_definitions(faderOS-pico PRIVATE
    FADEROS_UART_INDEX=${FADEROS_UART_INDEX} FADEROS_UART_TX=${FADEROS_UART_TX}
    FADEROS_UART_RX=${FADEROS_UART_RX} FADEROS_PANEL_BAUD=${FADEROS_PANEL_BAUD}
    PICO_STDIO_USB_STDOUT_TIMEOUT_US=0)
  pico_enable_stdio_uart(faderOS-pico 0)
  pico_enable_stdio_usb(faderOS-pico 1)
  pico_add_extra_outputs(faderOS-pico)
endif()

# The control model can be built/tested without any desktop adapter dependencies.
if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
  add_executable(link-tests test/link/core_test.cpp)
  target_link_libraries(link-tests PRIVATE picohost_core)
  target_compile_options(link-tests PRIVATE -UNDEBUG)
  add_test(NAME link-tests COMMAND link-tests)
  add_executable(lcd-graphics-tests test/link/lcd_graphics_test.cpp)
  target_link_libraries(lcd-graphics-tests PRIVATE picohost_core)
  target_compile_options(lcd-graphics-tests PRIVATE -UNDEBUG)
  add_test(NAME lcd-graphics-tests COMMAND lcd-graphics-tests)
endif()
if(PICOHOST_BUILD_POSIX)
  find_package(Threads REQUIRED)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(CURL REQUIRED IMPORTED_TARGET libcurl>=7.86)
  find_package(OpenSSL REQUIRED)
  find_package(LibXml2 REQUIRED)
  find_package(nlohmann_json 3.2 REQUIRED)
  find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
  # Generate in the build tree; a clean checkout never needs generated assets.
  add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/generated/web_page.hpp
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/embed_web.py
      --output ${CMAKE_CURRENT_BINARY_DIR}/generated/web_page.hpp
    DEPENDS tools/embed_web.py web/index.html web/app.js web/styles.css VERBATIM)
  add_library(picohost_adapters STATIC src/link/obs_posix.cpp src/link/kavtor_posix.cpp
    src/link/vmix_posix.cpp src/link/atem_posix.cpp src/link/web_posix.cpp
    ${CMAKE_CURRENT_BINARY_DIR}/generated/web_page.hpp)
  target_include_directories(picohost_adapters PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)
  target_link_libraries(picohost_adapters PUBLIC picohost_core Threads::Threads
    PkgConfig::CURL OpenSSL::Crypto LibXml2::LibXml2 nlohmann_json::nlohmann_json)
  add_executable(picohost-link src/link/main_posix.cpp src/link/baud_posix.cpp)
  set_target_properties(picohost-link PROPERTIES OUTPUT_NAME faderOS)
  # Preserve existing launchers while publishing the executable under its product name.
  add_custom_command(TARGET picohost-link POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:picohost-link>
      $<TARGET_FILE_DIR:picohost-link>/picohost-link VERBATIM)
  target_compile_options(picohost-link PRIVATE -fno-exceptions -fno-rtti)
  target_link_libraries(picohost-link PRIVATE picohost_adapters)
  install(TARGETS picohost-link RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
  install(PROGRAMS ${CMAKE_CURRENT_BINARY_DIR}/picohost-link DESTINATION ${CMAKE_INSTALL_BINDIR})
  install(DIRECTORY examples/ DESTINATION ${CMAKE_INSTALL_DATADIR}/faderOS/examples)
  if(BUILD_TESTING)
    execute_process(COMMAND ${Python3_EXECUTABLE} -c "import websockets"
      RESULT_VARIABLE missing_websockets ERROR_QUIET)
    if(missing_websockets)
      message(FATAL_ERROR "Tests require Python websockets. Install test/requirements.txt or set BUILD_TESTING=OFF.")
    endif()
    function(host_test name source)
      add_executable(${name} ${source} ${ARGN})
      target_link_libraries(${name} PRIVATE picohost_adapters)
      # Existing tests use assert(), including operations with side effects.
      target_compile_options(${name} PRIVATE -UNDEBUG)
    endfunction()
    host_test(atem-transport-tests test/link/atem_transport_test.cpp)
    host_test(baud-posix-tests test/link/baud_posix_test.cpp src/link/baud_posix.cpp)
    host_test(baud-fallback-tests test/link/baud_fallback_test.cpp)
    host_test(adapter-router-tests test/link/adapter_router_test.cpp src/link/baud_posix.cpp)
    host_test(kavtor-supersource-tests test/link/kavtor_supersource_test.cpp)
    host_test(kavtor-next-tests test/link/kavtor_next_test.cpp)
    host_test(kavtor-manual-tests test/link/kavtor_manual_test.cpp)
    foreach(probe obs kavtor vmix atem web transitions dsk outputs tbar servers)
      host_test(${probe}-probe test/link/${probe}_probe.cpp)
    endforeach()
    foreach(name atem-transport-tests baud-posix-tests baud-fallback-tests adapter-router-tests kavtor-next-tests kavtor-manual-tests kavtor-probe kavtor-supersource-tests)
      add_test(NAME ${name} COMMAND ${name})
    endforeach()
    foreach(name obs vmix atem atem_dsk web web_atem transitions dsk outputs tbar servers)
      add_test(NAME ${name} COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test/link/${name}_test.py)
      set_tests_properties(${name} PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
        ENVIRONMENT "PICOHOST_TEST_BUILD_DIR=${CMAKE_CURRENT_BINARY_DIR}" TIMEOUT 180)
    endforeach()
    find_program(FADEROS_NODE_EXECUTABLE NAMES node nodejs)
    if(FADEROS_NODE_EXECUTABLE)
      add_test(NAME web-kavtor-ui COMMAND ${FADEROS_NODE_EXECUTABLE} test/link/web_kavtor_test.js)
      set_tests_properties(web-kavtor-ui PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})
    else()
      message(STATUS "Node.js not found: skipping JavaScript web UI tests")
    endif()
    add_test(NAME cli-help COMMAND picohost-link --help)
    add_test(NAME cli-version COMMAND picohost-link --version)
  endif()
endif() # PICOHOST_BUILD_POSIX
find_package(Doxygen QUIET)

  set(DOXYGEN_EXTRACT_ALL YES)
  set(DOXYGEN_GENERATE_LATEX NO)
  doxygen_add_docs(docs src/link README.md COMMENT "Generate API documentation")
